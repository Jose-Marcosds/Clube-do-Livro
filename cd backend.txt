@app.get("/api/livros")
def listar_livros():
    conn = get_db_connection()
    cursor = conn.cursor()
    livros = cursor.execute("SELECT * FROM livros ORDER BY id DESC").fetchall()
    conn.close()
    return [dict(livro) for livro in livros]
